+  check_eq "$1" 0 "$?"
+}
+inv_eq() {

+run_expect() {

+  shift 3

+  check_eq "$label rc" "$want_rc" "$rc"
+  check_eq "$label stdout empty" 0 "$(bytes "$tmp/out")"
